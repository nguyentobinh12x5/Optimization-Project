\section{Motivation}
% ============================================================

\begin{frame}{The problem, in plain words}
  An investor has \textbf{1 unit of capital} and must split it across \textbf{98 of the 100
  stocks} in the VN100 basket (2 dropped for insufficient trading history), using 4 years of
  daily prices. What fraction of capital should go into each stock? This is the general
  problem of \textbf{portfolio optimization}.

  \vspace{0.3cm}
  \textbf{Markowitz mean--variance is the textbook starting point} --- but three questions
  decide what \emph{else} a general formula should be able to switch on, not what it must
  include unconditionally:
  \begin{enumerate}
    \item<2-> \textbf{Are the estimates trustworthy?} A sample mean return is a bad estimator
          over a short window --- should the formula be able to hedge against that?
    \item<3-> \textbf{Is holding 98 stocks worth it?} Every position costs money to open,
          monitor, and close --- should the formula be able to prefer fewer names?
    \item<4-> \textbf{Risk still matters}, in every configuration --- the one term that always
          stays on.
  \end{enumerate}

  \vspace{0.3cm}
  \onslide<5->{\centering \textit{Whether each option actually helps is not assumed here ---
  it is tested empirically on VN100 later, and the answer is not the obvious one.}}
\end{frame}

\section{The Model}
% ============================================================

\begin{frame}{Warm-up: what is $w$, concretely?}
  \textbf{$w$ is the decision variable} --- one number per stock, the \emph{fraction of
  capital} put into it.
  \vspace{0.12cm}
  \textbf{Toy example, 3 stocks:} split capital across A, B, C:
  \[
    w_A = 0.3, \quad w_B = 0.3, \quad w_C = 0.4, \qquad w_A+w_B+w_C = 1
  \]
  30\% of capital in A, 30\% in B, 40\% in C --- \textbf{always summing to 1} (all the capital
  gets used, none left idle).
  \vspace{0.12cm}
  If each stock's expected return is $\muhat_A=10\%$, $\muhat_B=5\%$, $\muhat_C=8\%$, this
  portfolio's expected return is just a weighted average:
  \[
    w^\top\muhat = 0.3(10\%) + 0.3(5\%) + 0.4(8\%) = 7.7\%
  \]
  \vspace{0.12cm}
  \textbf{Simplest possible question:} choose $w$ to make return as large as possible ---
  \[
    \max_{w}\ w^\top\muhat \quad \text{s.t.}\quad w_A+w_B+w_C=1
  \]
  \small The obvious (and naive) answer: put \textbf{100\% into A} (highest $\muhat$) --- but
  that ignores risk entirely. Fixing that is exactly what the full model does next.
\end{frame}

\begin{frame}{Four objectives $\to$ four terms}
  \begin{equation}\label{eq:main}
    \min_{w \in \R^{98}} \quad
      \underbrace{-\muhat^\top w}_{\text{return}}
      \;+\; \underbrace{\kappa \norm{\Ssqrt w}_2}_{\text{robustness}}
      \;+\; \underbrace{\gamma\, w^\top \Sig w}_{\text{variance}}
      \;+\; \underbrace{\lambda \norm{w}_1}_{\text{sparsity}}
    \quad \text{s.t.} \quad \one^\top w = 1
  \end{equation}
  \small \textit{Formulation follows Chen, Ahipa\c{s}ao\u{g}lu, Zhang \& Yang (2024),
  arXiv:2412.19462 --- adopted here as the single unifying model, not derived from scratch.}
  \vspace{0.2cm}
  \begin{table}
  \small
  \begin{tabular}{@{}ll@{}}
  \toprule
  \textbf{Term} & \textbf{Role} \\
  \midrule
  $-\muhat^\top w$ & maximize expected return \\
  $\kappa \norm{\Ssqrt w}_2$ & worst case if $\muhat$ is wrong (derived, not assumed) \\
  $\gamma\, w^\top \Sig w$ & classical Markowitz variance penalty \\
  $\lambda \norm{w}_1$ & convex surrogate for \# holdings; also caps gross leverage \\
  \bottomrule
  \end{tabular}
  \end{table}
  \textbf{No $w \ge 0$} here: short selling is allowed in the general model --- a deliberate
  design choice, revisited for the long-only variants later.
\end{frame}

\begin{frame}{Type of problem $\Rightarrow$ how we solve it}
  \small
  \textbf{Convex}: each term is affine, a norm, or a PSD quadratic, and the feasible set is
  affine (full term-by-term proof: backup) $\Rightarrow$ \textbf{every local optimum is
  global.} So a \emph{local}, first-order iteration is enough --- convexity promotes whatever
  it converges to into the global optimum.

  \vspace{0.15cm}
  But \cref{eq:main} is \textbf{nonsmooth}: $\kappa\norm{\Ssqrt w}_2$ at $\Ssqrt w=0$, and
  $\lambda\norm{w}_1$ wherever some $w_i=0$ --- i.e.\ exactly where the solution lives.
  Plain gradient descent does not apply.

  \vspace{0.15cm}
  \textbf{Our approach: split $f=g+h$, then alternate two easy moves each iteration.}
  \[
    f(w) = \underbrace{-\muhat^\top w + \kappa\norm{\Ssqrt w}_2 + \gamma w^\top\Sig w}
           _{g:\ \text{smoother --- subgradient step}}
         + \underbrace{\lambda\norm{w}_1 + I_{\{\one^\top w=1\}}(w)}_{h:\ \text{nonsmooth --- prox step}}
  \]
  \begin{enumerate}
    \item \textbf{Step down on $g$:} $z = w_k - \alpha_k v_k$, $v_k\in\partial g(w_k)$,
          $\alpha_k=\alpha_0/\sqrt{k+1}$ --- an ordinary gradient move, $h$ ignored for now.
    \item \textbf{Snap back with $h$:} $w_{k+1}=\mathrm{prox}_{\alpha_k h}(z)$ --- pulls $z$ onto
          a point that is sparse \emph{and} on budget ($\one^\top w=1$), at once (next slide).
  \end{enumerate}
  The budget constraint lives \emph{inside} $h$, folded into step 2 --- not handled separately.
\end{frame}

\begin{frame}{The prox step: why it must be solved jointly}
  \small
  Need $w$ close to $z$, sparse ($\ell_1$ pulls small entries to $0$), and on budget
  ($\one^\top w=1$) --- \emph{at the same time}, not sparsify first and patch the budget after:
  \[
    \mathrm{prox}_{th}(z) = \arg\min_w \tfrac12\norm{w-z}_2^2 + t\lambda\norm{w}_1
    \quad\text{s.t.}\quad \one^\top w = 1
  \]
  A Lagrange multiplier $\nu$ folds the budget constraint directly into the threshold, so both
  requirements are met together, coordinate by coordinate:
  \[
    w_i(\nu) = \mathcal{S}_{t\lambda}(z_i+\nu),
    \qquad \mathcal{S}_\tau(x)=\mathrm{sign}(x)\max(|x|-\tau,0)
  \]
  Pick $\nu$ so that $\phi(\nu):=\sum_i \mathcal S_{t\lambda}(z_i+\nu)=1$. $\phi$ is continuous
  and nondecreasing $\Rightarrow$ \textbf{bisection} (tol $10^{-12}$) finds it fast. Coordinates
  with $|z_i+\nu^\star|\le t\lambda$ land on \good{exactly zero} --- true sparsity, not
  near-zero.
\end{frame}

\section{One Formula, Six Methods}
% ============================================================

\begin{frame}{From one formula to six named strategies}
  \footnotesize
  Each variant tested later is \cref{eq:main} with certain terms fixed or dropped --- no
  variant introduces anything not already in the general objective.
  \vspace{0.15cm}
  \begin{table}
  \centering
  \tiny
  \begin{tabular}{@{}llccl@{}}
  \toprule
  \textbf{Method} & \textbf{Type} & $\kappa$ & $\lambda$ & \textbf{Objective (from \cref{eq:main})} \\
  \midrule
  A: Robust Markowitz    & long-only ($w\ge0$)       & searched  & $\equiv0$ &
    $-\muhat^\top w+\kappa\norm{\Ssqrt w}_2+\gamma w^\top\Sig w$ \\
  B: Classical Markowitz & long-only ($w\ge0$)       & $\equiv0$ & $\equiv0$ &
    $-\muhat^\top w+\gamma w^\top\Sig w$ \\
  C: Equal-weight $1/N$  & no optimization           & --        & --        & -- \\
  D: VN100 buy\&hold     & no optimization           & --        & --        & -- \\
  E: Sparse-only         & long-short ($w\gtrless0$) & $\equiv0$ & searched  &
    $-\muhat^\top w+\gamma w^\top\Sig w+\lambda\norm{w}_1$ \\
  F: Full equation       & long-short ($w\gtrless0$) & searched  & searched  &
    $-\muhat^\top w+\kappa\norm{\Ssqrt w}_2+\gamma w^\top\Sig w+\lambda\norm{w}_1$ \\
  \bottomrule
  \end{tabular}
  \end{table}
  \vspace{0.15cm}
  \footnotesize
  \textbf{Long-only} ($w_i\ge0$ for all $i$): can only buy/hold each stock --- no short selling.
  \textbf{Long-short} ($w_i\gtrless0$, no sign restriction): $w_i>0$ means long (buy), $w_i<0$
  means short (bet the price falls) --- only E/F allow this; A--D do not.
  \vspace{0.1cm}
  A/B add $w\ge0$: the simplex constraint then forces $\norm{w}_1\equiv1$, so $\lambda$ is
  dropped from their search rather than tuned pointlessly --- any concentration A/B show is a
  \emph{free} effect of the simplex geometry, not a designed sparsity term.
\end{frame}

\section{Walk-Forward Backtest \& Results}
% ============================================================

\begin{frame}{Walk-forward out-of-sample backtest}
  \textbf{Why:} in-sample results above optimize on data already seen --- say nothing about
  deployed performance.

  \vspace{0.2cm}
  \begin{itemize}
    \item Rolling \textbf{24-month} window: 18 months fit + 6 months validate, sliding 1 month
          at a time (diagram next)
    \item \textbf{Monthly} rebalance, \textbf{494 out-of-sample days}, \textbf{25 rebalances}
    \item Ledoit--Wolf shrinkage; \textbf{0.20\%} transaction cost per unit turnover
    \item Two benchmarks built from the \emph{same} 98-ticker universe (not an external VN30
          index): C (equal-weight, monthly) and D (VN100 buy\&hold, once)
  \end{itemize}
\end{frame}

  \vspace{0.1cm}
  Within each period, four steps run in order, for every variant A--F:
  \begin{enumerate}
    \item \textbf{Fit} every hyperparameter candidate in the grid (e.g.\
          $(\kappa,\gamma)\in\{0,0.5,1,2\}\times\{1,5,10\}$ for A) on $E$.
    \item \textbf{Score} each candidate by \textbf{Sharpe on $V$} --- data its fit never saw.
    \item \textbf{Keep} the winning $(\kappa^\star,\gamma^\star)$.
    \item \textbf{Refit and deploy}: re-estimate on the full $E\cup V$, re-solve at the winning
          hyperparameters --- that $w^\star$ buys the single
          \textcolor{green!50!black}{\textbf{deployed}} month.
  \end{enumerate}
  \vspace{0.05cm}
  The window then \textbf{slides forward by exactly 1 month} and repeats --- 25 times overall.
  Period 1 and Period 2 already share \textbf{23 of 24} months of data --- yet $(\kappa,\gamma)$
  still swings between periods (backup: full instability plot). A 6-month validation slice is a
  noisy yardstick even when the underlying data barely changes.

  \vspace{0.1cm}
  \textbf{Why:} $\gamma$ is auto-selected each period to maximize validation Sharpe --- if one
  ticker's estimated risk-adjusted profile dominates that window, nothing stops the optimizer
  from betting the whole book on it. As an investor I want my portfolio to hold around
  \textbf{5 names}, not swing to 1 $\Rightarrow$ needs an explicit constraint (next slide).
\end{frame}

\begin{frame}{A constraint to fix it: results}
  \textbf{Fix:} add $w_i \le 1/K$ for A/B ($K$ = investor-chosen min.\ names held, genuinely
  convex). $K=5$ gives a 20\% per-name cap.
  \vspace{0.2cm}
  \small
  \begin{table}
  \begin{tabular}{@{}lrrrrr@{}}
  \toprule
  \textbf{Method} & CumRet & AnnRet & Vol & Sharpe & MaxDD \\
  \midrule
  A, uncapped      & 37.7\% & 17.7\% & 30.9\% & 0.68 & $-39.7\%$ \\
  A, capped $20\%$ & \good{56.8\%} & \good{25.8\%} & \good{21.3\%} & \good{1.19} & \good{$-25.0\%$} \\
  B, uncapped      & 74.2\% & 32.7\% & 31.6\% & 1.05 & $-39.7\%$ \\
  B, capped $20\%$ & \good{76.3\%} & \good{33.6\%} & \good{22.8\%} & \good{1.39} & \good{$-25.0\%$} \\
  \bottomrule
  \end{tabular}
  \end{table}
  \small The cap \emph{raises} return while \emph{cutting} vol/drawdown by a third ---
  \good{long-only beats both benchmarks}; long-short without a leverage bound (E/F) is a
  cautionary tale the other way (backup: mechanism \& worst day).
\end{frame}

\begin{frame}{Turnover and transaction cost --- worked example}
  \small
  At each rebalance, cost is computed from the \textbf{full} weight vector, compared against
  the \textbf{drifted} portfolio just before rebalancing --- not last period's target:
  \[
    \text{turnover} = \sum_{i=1}^{N} \bigl|w^{\text{new}}_i - w^{\text{prev, drifted}}_i\bigr|
    \qquad
    \text{cost} = \text{fee} \times \text{turnover}
  \]
  \vspace{0.15cm}
  \textbf{Example} (2 of the 98 tickers shown; every other ticker unchanged $\Rightarrow$
  contributes $0$):
  \begin{center}
  \begin{tabular}{@{}lcc@{}}
  \toprule
   & Ticker A & Ticker B \\
  \midrule
  Drifted weight, just before rebalance & 60\% & 40\% \\
  New target weight, this period        & 30\% & 70\% \\
  $|\Delta w_i|$                        & 30\% & 30\% \\
  \bottomrule
  \end{tabular}
  \end{center}
  \[
    \text{turnover} = |30\%-60\%| + |70\%-40\%| = 30\%+30\% = \good{60\%}
    \qquad
    \text{cost} = 0.20\% \times 60\% = \good{0.12\%}
  \]
  Deducted from the \textbf{first day's} return of the new holding period only.

  \textbf{Design choice:} cost is a friction \emph{after} $w^\star$ is chosen ---
  \cref{eq:main} has no $w_{\text{prev}}$ term; a penalty $\eta\norm{w-w_{\text{prev}}}_1$ could
  be added but would be a 4th hyperparameter on an already-unstable window --- kept external to
  keep the concentration cap's attribution clean.
\end{frame}

\section{Conclusion}
% ============================================================

\begin{frame}{Conclusion}
  \begin{itemize}
    \item Formulated a \textbf{convex} model $\Rightarrow$ every local minimum \textbf{is} the
          global minimum. The robust term is \textbf{derived}, not assumed; $\lambda\norm{w}_1$
          does double duty --- sparsity \emph{and} a leverage cap.
    \item Solved with a \textbf{proximal-subgradient method}, whose prox step handles the
          $\ell_1$ penalty and the budget constraint \emph{jointly and exactly} (bisection on a
          scalar multiplier) so soft-thresholded zeros survive.
    \item \textbf{Out-of-sample}: tested on the most recent \textbf{4 years} of data from
          \textbf{VN100} (Vietnam's 100 largest listed stocks). Results hold for \emph{this}
          dataset --- a different market or period may change the outcome.
    \item Long-short \textbf{without a leverage cap} (methods E/F in this study) suffered
          \textbf{total capital loss} --- categorically worse, not just riskier: the sharpest
          empirical lesson of the study.
  \end{itemize}
\end{frame}

\begin{frame}{Decision variable and constants}
  \textbf{Decision variable:} $w \in \R^{98}$ --- fraction of capital in each stock
  ($w_i < 0$ = short).

  \vspace{0.3cm}
  \textbf{Constants} (estimated once, then fixed):
  \begin{itemize}
    \item $\muhat \in \R^{98}$ --- sample mean daily return
    \item $\Sig \in \mathbb{S}^{98}_+$ --- sample covariance
    \item $\Ssqrt$ --- symmetric PSD square root, $\Ssqrt\Ssqrt = \Sig$, via
          \textbf{eigendecomposition} (not Cholesky: $\Sig$ is only PSD, with a few
          numerically negative eigenvalues to clip)
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Hyperparameters} $\kappa,\gamma,\lambda \ge 0$: \emph{not} decision variables ---
  chosen by us (\S~Results) or by validation (backtest).
\end{frame}

\begin{frame}{Why is this \emph{min} convex? Term by term}
  \[
    \min_{w \in \R^{98}} \quad
      \underbrace{-\muhat^\top w}_{\text{(a) affine}}
      \;+\; \underbrace{\kappa \norm{\Ssqrt w}_2}_{\text{(b) norm}}
      \;+\; \underbrace{\gamma\, w^\top \Sig w}_{\text{(c) PSD quadratic}}
      \;+\; \underbrace{\lambda \norm{w}_1}_{\text{(d) norm}}
    \quad \text{s.t.} \quad \one^\top w = 1
  \]
  \small
  \begin{itemize}
    \item \textbf{(a)} $-\muhat^\top w$ is \textbf{linear in $w$} --- linear functions are
          convex (and concave).
    \item \textbf{(b)} $\kappa\norm{\Ssqrt w}_2$ is a \textbf{norm} of a linear map of $w$
          --- every norm is convex, by the triangle inequality.
    \item \textbf{(c)} $\gamma\, w^\top\Sig w$ is a \textbf{quadratic form with
          $\Sig\succeq0$} (PSD, a covariance matrix) --- convex quadratics are exactly the PSD
          ones.
    \item \textbf{(d)} $\lambda\norm{w}_1$ is a norm too --- convex, same reason as (b).
    \item The feasible set $\{w:\one^\top w=1\}$ (and $w\ge0$ for A/B) is
          \textbf{affine/polyhedral} --- convex.
  \end{itemize}
  \begin{center}
  \large A \textbf{nonnegative sum} of convex functions is convex
  $\Rightarrow$ the whole objective is convex.
  \end{center}
\end{frame}

\begin{frame}{Long-short without a leverage cap: E and F wiped out}
  E and F keep \cref{eq:main} exactly as stated --- short positions allowed, no bound on
  $\norm{w}_1$ beyond the soft $\lambda$ penalty.

  \vspace{0.2cm}
  \textbf{F's worst day (2025-06-02):} no individual ticker moved more than $\pm7\%$, yet a
  portfolio of \textbf{49 positions above 100\% of capital each} (gross exposure
  \bad{$133\times$}) lost \bad{39\%} in that one session.

  \vspace{0.2cm}
  A few such sessions inside an un-rebalanced month are enough to breach $-100\%$ --- the
  simulator floors the loss and marks the rest of the month at $0\%$. Recurred for both E and F,
  independently, across periods.

  \vspace{0.3cm}
  \begin{center}
  \Large \bad{Negative result:} a per-position cap (effective for A/B) would \emph{not} fix
  this --- the pathology is \emph{total} exposure, not concentration in one name.
  \end{center}
\end{frame}

\begin{frame}{Interpretation}
  \begin{itemize}
    \item \textbf{The robust term does not earn its keep, in or out of sample.} B's Sharpe
          ($1.39$) beats A's ($1.19$); A's own search picks $\kappa=0$ in 18 of 25 periods.
    \item \textbf{Both long-only variants beat both benchmarks} --- but not for free: B's
          fees (2.15\%, 42.9\%/month turnover) and A's (3.00\%, 60.1\%) run 4--6$\times$ C's,
          driven by $(\kappa,\gamma)$ still swinging between adjacent months on a noisy 6-month
          validation window.
    \item \textbf{The long-short extension is the sharpest lesson.} E/F are not merely
          worse than A--D, they are \emph{categorically} different: nothing in their
          formulation bounds total leverage the way the long-only constraint set implicitly
          does.
  \end{itemize}
  \vspace{0.3cm}
  \centering \textit{A real, evidenced edge from the concentration cap --- and a cautionary
  tale from removing it.}
\end{frame}

\begin{frame}{Limitations}
  \small
  \begin{itemize}
    \item Survivorship bias: current VN100 membership applied backward
    \item $\muhat$ remains a poor estimator --- the robust term does not blunt this; the
          concentration cap does, by making single-ticker bets infeasible
    \item No prediction model: $\muhat$ is always the trailing sample mean, never a forecast ---
          a predict-then-optimize benchmark (predictor $\to$ same \cref{eq:main}) is a genuinely
          separate follow-on study, not attempted here
    \item Unconstrained leverage is a distinct, more severe failure mode than concentration ---
          a leverage cap $\norm{w}_1\le L$ is the natural fix, left as future work
    \item OOS window is short: $\sim$24 months, one market regime
    \item 6-month validation window causes mild parameter overfitting
    \item Single historical path --- no bootstrap confidence intervals
    \item Flat transaction-cost model, no slippage/market impact/margin calls
  \end{itemize}
\end{frame}
